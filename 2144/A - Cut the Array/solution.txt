t=int(input())
for _ in range(t):
    n=int(input())
    l=list(map(int,input().split()))
    verif=False
    i=1
    j=1
    for i in range(1,n-1):
        for j in range(1,n-1):
            if i+j<=n-1:
                a1=l[:i]
                s1=sum(a1)
                a2=l[i:i+j]
                s2=sum(a2)
                a3=l[i+j:]
                s3=sum(a3)
                if s1%3==s2%3==s3%3 or (s1%3!=s2%3 and s2%3!=s3%3 and s1%3!=s3%3):
                    verif=True
            break
        break
    if verif:
        print(i, i+j)
    else:
        print(0, 0)
